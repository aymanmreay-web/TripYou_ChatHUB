# سكريبت نشر السيرفر المباشر — TripYou_ChatHUB
# تم حذف وإيقاف Chatwoot نهائياً وتحرير كافة الموارد على السيرفر (HUB-009)

# ============ 1. الانتقال للمجلد المعتمد ============
cd /opt/resvbar-hub

# ============ 2. ضبط المستودع الرسمي الوحيد ============
git remote set-url origin https://github.com/aymanmreay-web/TripYou_ChatHUB.git

# ============ 3. جلب آخر كود مبني ============
git pull origin main

# ============ 4. بناء وتشغيل الحاوية بأحدث كود ============
docker compose -f docker-compose.hub.yml build --no-cache
docker compose -f docker-compose.hub.yml up -d

# ============ 5. فحص حالة الحاويات ============
docker ps --filter "name=tripyou-chathub-web"
